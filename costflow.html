คู่มือเชื่อมต่อ Google Sheet ใน 4 ขั้นตอน (Apps Script)

ทำตามขั้นตอนนี้เพื่อใช้งานระบบบันทึกแบบเรียลไทม์ได้ด้วยตนเอง

1

เปิดระบบพัฒนา Google Apps Script

เข้าสู่ Google Sheet ของคุณ จากนั้นไปที่แถบเมนูด้านบน เลือก ส่วนขยาย (Extensions) > Apps Script

2

คัดลอกโค้ดด้านล่างนี้ไปวาง

ลบโค้ดเดิมทั้งหมดในแผงเขียนโค้ดและวางสคริปต์นี้ลงไปแทนที่:

/* CostFlow Google Apps Script Integrator */
function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var header = data[0];
  var rows = [];
  
  for (var i = 1; i < data.length; i++) {
    var row = {};
    for (var j = 0; j < header.length; j++) {
      row[header[j]] = data[i][j];
    }
    rows.push(row);
  }
  
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var rawData = e.postData.contents;
    var payload = JSON.parse(rawData);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // Clear existing data (keep header)
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).clearContent();
    }
    
    // Header check / creation
    var headers = ["รายการ", "หมวดหมู่", "จำนวน", "ราคาต่อหน่วย", "ราคารวม", "หมายเหตุ"];
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    
    if (payload && payload.length > 0) {
      var writeData = [];
      for (var i = 0; i < payload.length; i++) {
        var item = payload[i];
        writeData.push([
          item["รายการ"] || "",
          item["หมวดหมู่"] || "",
          Number(item["จำนวน"]) || 0,
          Number(item["ราคาต่อหน่วย"]) || 0,
          Number(item["ราคารวม"]) || 0,
          item["หมายเหตุ"] || ""
        ]);
      }
      sheet.getRange(2, 1, writeData.length, headers.length).setValues(writeData);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: payload.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
3

ทำการ Deploy เป็นเว็บแอปพลิเคชัน (Web App)

คลิกที่ปุ่ม การทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้ใหม่ (New deployment) เลือกประเภทเป็น เว็บแอป (Web app) กำหนดให้ผู้มีสิทธิ์เข้าถึงเป็น "ทุกคน" (Anyone) จากนั้นกดปุ่มแชร์สิทธิ์

4

บันทึก URL ลงในเว็บแอปนี้
